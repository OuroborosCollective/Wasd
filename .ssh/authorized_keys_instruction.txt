# SSH Security Hardening Documentation
# Target: .ssh/authorized_keys_instruction.txt

# 1. Create dedicated deploy user 'ouroboros'
# Command: sudo adduser --disabled-password --gecos "" ouroboros

# 2. Setup SSH directory and permissions
# Command: sudo mkdir -p /home/ouroboros/.ssh
# Command: sudo touch /home/ouroboros/.ssh/authorized_keys
# Command: sudo chown -R ouroboros:ouroboros /home/ouroboros/.ssh
# Command: sudo chmod 700 /home/ouroboros/.ssh
# Command: sudo chmod 600 /home/ouroboros/.ssh/authorized_keys

# 3. SSH Public Keys
# Add authorized public keys to: /home/ouroboros/.ssh/authorized_keys

# 4. SSH Daemon Hardening
# File: /etc/ssh/sshd_config
# Ensure the following settings are active:
# ----------------------------------------
# PermitRootLogin prohibit-password
# PasswordAuthentication no
# PubkeyAuthentication yes
# ChallengeResponseAuthentication no
# UsePAM yes

# 5. Validation and Restart
# Command: sudo sshd -t && sudo systemctl restart sshd
