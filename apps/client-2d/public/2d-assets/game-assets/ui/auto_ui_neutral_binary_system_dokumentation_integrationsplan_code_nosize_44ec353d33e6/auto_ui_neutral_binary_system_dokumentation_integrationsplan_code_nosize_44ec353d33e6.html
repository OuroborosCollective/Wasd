code Integration Panel: SymbolRenderer.tsx 

import { Container, Sprite, Graphics } from 'pixi.js';
import { IconData, RarityToken } from '@arelorian/core';

// Eldritch Modular Gothic Renderer
export const RenderArelorianIcon = (data: IconData) => {
  const container = new Container();
  
  // Apply Rarity Glow (Sunset Gold / Mythic Red)
  const glow = new Graphics();
  glow.beginFill(data.rarityColor, 0.4);
  glow.drawCircle(32, 32, 28);
  glow.filters = [new BlurFilter(15)];
  
  const iconSprite = Sprite.from(data.assetId);
  iconSprite.anchor.set(0.5);
  iconSprite.scale.set(0.8);

  container.addChild(glow, iconSprite);
  return container;
};
