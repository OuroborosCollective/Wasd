  updateResonance(resonance: Record<string, { faith: number; aggression: number; curiosity: number }>): void {
    let changed = false;
    for (const [key, val] of Object.entries(resonance)) {
      const existing = this.resonanceByChunk.get(key);
      if (!existing || existing.faith !== val.faith || existing.aggression !== val.aggression || existing.curiosity !== val.curiosity) {
        this.resonanceByChunk.set(key, val);
        changed = true;
        // If chunk is already loaded, we might want to "re-mutate" it,
        // but for now, we'll only use it for new generations or manual reloads.
      }
    }
